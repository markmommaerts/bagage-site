  /* Revidd-tags omzetten naar thema's, subthema's, bestemmingen, rijen en prijs */
  function classify(item) {
    const themes = new Set(), subthemes = new Set(), destinations = new Set(), collections = new Set();
    let price = '', partner = '';
    tagNames(item).forEach(name => {
      const key = name.toLocaleUpperCase('nl').replace(/^\(\s*|\s*\)$/g, '');
      if (!key || IGNORED_TAGS.includes(key)) return;
      const priceMatch = key.match(PRICE_TAG);
      if (priceMatch) { price = formatPrice(priceMatch[1], priceMatch[2]); return; }
      if (PAID_TAGS.includes(key)) { price = price || 'Betalend'; return; }
      const partnerMatch = key.match(PARTNER_TAG);
      if (partnerMatch) {
        const name = partnerMatch[1].trim();
        partner = name.includes(' ') ? name : name.replace(/-/g, ' ');   // IN-FLANDERS-FIELDS → IN FLANDERS FIELDS
        return;
      }
      if (THEME_TAGS[key]) themes.add(THEME_TAGS[key]);
      else if (SUBTHEME_TAGS[key]) { themes.add(SUBTHEME_TAGS[key][0]); subthemes.add(SUBTHEME_TAGS[key][1]); }
      else if (COLLECTION_TAGS[key]) collections.add(COLLECTION_TAGS[key]);
      else destinations.add(DESTINATION_NAMES[key] || titleCase(key));
    });
    return { themes:[...themes], subthemes:[...subthemes], destinations:[...destinations], collections:[...collections], price, partner };
  }

  /* Welke pagina hoort bij een item: eigen tag/lijst, anders die van zijn seizoen of reeks */
  function buildPageMap(items) {
    const byId = Object.fromEntries(items.map(i => [i.id, i]));
    const parent = {};
    items.forEach(s => (s.seasons || []).forEach(se => { parent[se.id] = s.id; }));
    const own = {};
    Object.entries(SERIES_PAGES).forEach(([p, ids]) => ids.forEach(id => { own[id] = p; }));
    items.forEach(i => tagNames(i).forEach(n => {
      const m = n.trim().match(PAGE_TAG);
      if (m) own[i.id] = m[1].trim().toLocaleLowerCase('nl').replace(/^\/?(?:[a-z]{2}\/)?(?:pages\/)?/, '').replace(/\s+/g, '-');
    }));
    const memo = {};
    const find = (id, depth) => {
      if (!id || depth > 4) return '';
      if (id in memo) return memo[id];
      const it = byId[id];
      return memo[id] = own[id] || (it && (find(it.seasonId, depth + 1) || find(it.seriesId, depth + 1))) || find(parent[id], depth + 1) || '';
    };
    pageOf = id => find(id, 0);
  }

  /* ---------- Kaart opbouwen ---------- */
  /* linkItem: het item waarnaar de kaart linkt. Voor een reeks is dat het eerste
     seizoen, want daar kan de kijker de afleveringen afspelen. */
  function renderCard(item, episodes, seasonCount, linkItem = item) {
    const isSeries = item.type === 'RELATIONSHIP';
    const info = classify(item);
    const page = pageOf(item.id);
    const href = page ? pageHref(page) : STORE + linkItem.seoSlug + '-' + linkItem.id;
    const description = (item.description || '').trim();

    const card = el('article', 'card');
    card.dataset.format = isSeries ? 'reeks' : 'documentaire';
    card.dataset.themes = info.themes.join('|');
    card.dataset.subthemes = info.subthemes.join('|');
    card.dataset.destinations = info.destinations.join('|');
    card.dataset.collections = info.collections.join('|');
    card.dataset.sortTitle = sortTitle(item.title);
    card.dataset.published = Date.parse(item.publishOn || item.createdAt) || 0;
    card.dataset.page = page;
    if (tagNames(item).some(n => n.trim().toLocaleUpperCase('nl') === LEAD_TAG)) card.dataset.lead = '1';
    /* Prijs uit de tag; staat de video in Revidd zelf als betalend, dan minstens "Betalend" */
    const price = info.price || (item.isFree === false ? 'Betalend' : '');
    card.dataset.free = price ? '' : '1';

    /* Poster */
    const poster = el('a', 'poster');
    /* De catalogus staat in Revidd in een ingesloten kader (iframe).
       target _top opent de video als volledige pagina i.p.v. binnen dat kader. */
    poster.href = href;
    poster.target = '_top';
    const img = el('img');
    img.src = imageUrl(item.coverImagePortrait) || imageUrl(item.coverimage);
    img.alt = item.title.trim();
    img.loading = 'lazy';
    poster.append(img);
    if (info.partner) {
      const band = el('div', 'partner');
      band.append(el('small', '', PARTNER_LABEL), el('b', '', info.partner));
      poster.append(band);
    }
    const hover = el('div', 'hover');
    hover.append(el('p', '', description), el('span', '', 'BEKIJK →'));
    poster.append(hover);

    /* Type + duur */
    const kindline = el('div', 'kindline');
    let length = '';
    if (isSeries) {
      const durations = episodes.map(e => e.actualmedia?.[0]?.duration).filter(Boolean);
      const avg = durations.length ? minutes(durations.reduce((a, b) => a + b, 0) / durations.length) : '';
      const count = episodes.length + (episodes.length === 1 ? ' aflevering' : ' afleveringen');
      const seasons = seasonCount > 1 ? seasonCount + ' seizoenen' : '';
      length = [avg, seasons, count].filter(Boolean).join(' · ');
    } else if (item.actualmedia?.[0]?.duration) {
      length = minutes(item.actualmedia[0].duration);
    }
    kindline.append(el('span', 'kind', isSeries ? 'REEKS' : 'DOCUMENTAIRE'));
    if (length) kindline.append(el('span', 'separator', '·'), el('em', '', length));

    /* Titel */
    const title = el('a', 'title', item.title.trim());
    title.href = href;
    title.target = '_top';

    card.append(poster, kindline, title);

    /* Thema */
    if (info.themes.length) card.append(el('div', 'theme', info.themes.join(' · ').toLocaleUpperCase('nl')));
    if (price) card.append(el('div', 'price', price));

    if (description) card.append(el('p', 'mobile-desc', description));
    return card;
  }

  /* ---------- Subthema- en bestemmingskeuzes uit de gegevens ---------- */
  function buildChoices() {
    const present = key => new Set(cards.flatMap(card => values(card, key)));
    const subs = present('subthemes');
    const result = {};
    Object.values(SUBTHEME_TAGS).forEach(([theme, sub]) => {
      if (!subs.has(sub)) return;
      result[theme] = result[theme] || { key:'subthemes', multi:false, values:[] };
      if (!result[theme].values.includes(sub)) result[theme].values.push(sub);
    });
    const dests = [...present('destinations')].sort((a, b) => a.localeCompare(b, 'nl'));
    if (dests.length) result['Bestemmingen'] = { key:'destinations', multi:true, values:dests, grouped:true };
    return result;
  }

  /* Lege tegels verbergen: een thema verschijnt zodra er één video met die tag is */
  function hideEmptyTiles() {
    let visible = 0;
    buttons.forEach(button => {
      const f = button.dataset.filter;
      const has = cards.some(card => f === 'Bestemmingen'
        ? values(card, 'destinations').length > 0
        : values(card, 'themes').includes(f));
      button.hidden = !has;
      if (has) visible++;
    });
    filters.dataset.tiles = visible;
    filters.style.setProperty('--tiles', visible);
  }

  const themeMatch = (card, f) => f === 'Alles' || (f === 'Bestemmingen'
    ? values(card, 'destinations').length > 0
    : values(card, 'themes').includes(f));

  /* ---------- Startweergave: rijen ---------- */
  function buildRows(pool) {
    const sections = [];
    const newest = [...pool].sort(byNewest).slice(0, NEW_ROW_SIZE);
    if (newest.length) sections.push({ title:NEW_ROW_TITLE, items:newest });
    Object.values(COLLECTION_TAGS).forEach(name => {
      const items = pool.filter(card => values(card, 'collections').includes(name)).sort(byTitle);
      if (items.length) sections.push({ title:name, items });
    });
    buttons.filter(b => !b.hidden && b.dataset.filter !== 'Bestemmingen').forEach(button => {
      const items = pool.filter(card => themeMatch(card, button.dataset.filter)).sort(byTitle);
      if (items.length) sections.push({ title:button.dataset.filter, items,
        total:items.length, button });
    });

    rows.replaceChildren(...sections.map(section => {
      const block = el('section', 'row');
      const head = el('div', 'row-head');
      head.append(el('h2', '', section.title));
      const track = el('div', 'row-track');
      track.append(...section.items.map(card => {
        const copy = card.cloneNode(true);
        copy.hidden = false;
        return copy;
      }));
      if (section.button) {
        const more = el('button', 'row-more', 'ALLES BEKIJKEN (' + section.total + ') →');
        more.type = 'button';
        more.addEventListener('click', () => {
          section.button.click();
          filters.scrollIntoView({ block:'start', behavior:'smooth' });
        });
        head.append(more);
      }
      /* Pijltjes op computer om door de rij te bladeren */
      const nav = el('div', 'row-nav');
      const prev = el('button', '', '‹'), next = el('button', '', '›');
      prev.type = next.type = 'button';
      prev.setAttribute('aria-label', 'Vorige');
      next.setAttribute('aria-label', 'Volgende');
      prev.addEventListener('click', () => track.scrollBy({ left:-track.clientWidth }));
      next.addEventListener('click', () => track.scrollBy({ left:track.clientWidth }));
      const syncNav = () => {
        prev.disabled = track.scrollLeft <= 2;
        next.disabled = track.scrollLeft + track.clientWidth >= track.scrollWidth - 2;
      };
      track.addEventListener('scroll', syncNav, { passive:true });
      requestAnimationFrame(syncNav);
      nav.append(prev, next);
      head.append(nav);
      block.append(head, track);
      return block;
    }));
    return sections.length;
  }

  function applyFilter() {
    const conf = choices[filter];
    const inTheme = cards.filter(card => themeMatch(card, filter) && (!conf || selected.size === 0 ||
      values(card, conf.key).some(v => selected.has(v))) &&
      (filter !== 'Bestemmingen' || !continent ||
        values(card, 'destinations').some(v => continentOf(v) === continent)));

    /* Aantallen per soort binnen het gekozen thema; een lege soort wordt grijs */
    formatButtons.forEach(button => {
      const n = inTheme.filter(card => formatMatch(card, button.dataset.format)).length;
      button.querySelector('.n').textContent = n;
      button.disabled = n === 0 && button.dataset.format !== '';
    });
    /* Geen reeksen (of documentaires) in dit thema? Dan terug naar alles. */
    if (formatButtons.some(b => b.disabled && b.dataset.format === format)) {
      format = '';
      formatButtons.forEach(b => {
        b.classList.toggle('active', b.dataset.format === '');
        b.setAttribute('aria-pressed', String(b.dataset.format === ''));
      });
    }
    const shown = inTheme.filter(card => formatMatch(card, format));

    /* Zonder gekozen thema: rijen per thema. Met een thema: het volledige raster. */
    const rowView = filter === 'Alles';
    rows.hidden = !rowView;
    grid.hidden = rowView;
    sortBar.hidden = rowView;
    let count;
    if (rowView) {
      count = buildRows(shown);
    } else {
      rows.replaceChildren();
      cards.forEach(card => { card.hidden = !shown.includes(card); });
      grid.replaceChildren(...[...cards].sort(sort === 'new' ? byNewest : byTitle));
      count = shown.length;
    }
    empty.textContent = 'Geen verhalen gevonden voor deze selectie.';
    empty.hidden = count > 0;

    /* Gsm: zodra een thema gekozen is, verdwijnen de andere tegels en staan
       de subthema's en reportages meteen onder de keuze (CSS geldt enkel op gsm). */
    const focus = filter !== 'Alles';
    const wasFocused = root.classList.contains('is-focused');
    root.classList.toggle('is-focused', focus);
    if (focus !== wasFocused) {
      placePanel();
      const filtersTop = window.bagageTopOnPage
        ? window.bagageTopOnPage(filters) - window.bagageHeaderBottom()
        : filters.getBoundingClientRect().top;
      if (focus && mobile.matches && filtersTop < 0) {
        filters.scrollIntoView({ block:'start', behavior:'smooth' });
      }
    }
  }

  function syncOptions() {
    menu.querySelectorAll('[data-value]').forEach(item => {
      const active = selected.has(item.dataset.value);
      item.classList.toggle('active', active);
      item.setAttribute('aria-pressed', String(active));
    });
  }

  function optionButton(value, conf) {
    const option = document.createElement('button');
    option.type = 'button';
    option.textContent = value;
    option.dataset.value = value;
    option.setAttribute('aria-pressed', 'false');
    option.addEventListener('click', () => {
      if (selected.has(value)) {
        selected.delete(value);
      } else {
        if (!conf.multi) selected.clear();
        selected.add(value);
      }
      syncOptions();
      applyFilter();
    });
    return option;
  }

  function buildMenu() {
    menu.replaceChildren();
    const conf = choices[filter];
    panel.hidden = !conf;
    menu.hidden = !conf;
    menu.classList.toggle('grouped', !!(conf && conf.grouped));
    if (!conf) return;
    menu.setAttribute('aria-label', filter === 'Bestemmingen' ? 'Bestemmingen' : 'Subthema’s');
    if (conf.grouped) {
      /* Eerst alleen de werelddelen; na een keuze verschijnen de landen ervan */
      const present = [...Object.keys(CONTINENTS), 'Overige']
        .filter(c => conf.values.some(v => continentOf(v) === c));
      const continentMenu = el('div', 'continent-menu');
      present.forEach(c => {
        const option = el('button', '', c);
        option.type = 'button';
        option.dataset.continent = c;
        option.classList.toggle('active', c === continent);
        option.setAttribute('aria-pressed', String(c === continent));
        option.addEventListener('click', () => {
          continent = continent === c ? '' : c;   // nogmaals tikken = werelddeel loslaten
          selected.clear();
          buildMenu();
          applyFilter();
        });
        continentMenu.append(option);
      });
      menu.append(continentMenu);
      if (continent) {
        const group = el('div', 'dest-group');
        group.append(el('span', 'dest-label', 'LANDEN'),
          ...conf.values.filter(v => continentOf(v) === continent).map(value => optionButton(value, conf)));
        menu.append(group);
      }
    } else {
      conf.values.forEach(value => menu.append(optionButton(value, conf)));
    }
  }

  /* Bij het openen is geen tegel gekozen en staan de rijen per thema in beeld.
     Nogmaals op het gekozen thema tikken = deselecteren: terug naar de rijen. */
  buttons.forEach(button => button.addEventListener('click', () => {
    const prevFilter = filter;
    const deselect = button.classList.contains('active');
    filter = deselect ? 'Alles' : button.dataset.filter;
    selected.clear();
    continent = '';
    buttons.forEach(item => {
      const active = !deselect && item === button;
      item.classList.toggle('active', active);
      item.setAttribute('aria-pressed', String(active));
    });
    buildMenu();
    if (cards.length) applyFilter();
    placePanel();
    rememberTheme(prevFilter);
    backLinks.forEach(b => { b.hidden = filter === 'Alles'; });
  }));

  /* Terug naar het overzicht: thema uitzetten en naar de top van de catalogus */
  const backLinks = [...root.querySelectorAll('.back-overview')];
  backLinks.forEach(b => b.addEventListener('click', () => {
    const active = buttons.find(x => x.classList.contains('active'));
    if (active) active.click();
    root.scrollIntoView({ block:'start', behavior:'auto' });
  }));

  /* Gekozen thema in het adres bewaren (?thema=…), zodat 'terug' vanaf een titel weer bij dit thema uitkomt */
  let syncing = false;
  function rememberTheme(prevFilter) {
    if (syncing) return;
    try {
      const w = window.parent, u = new URL(w.location.href);
      if (filter === 'Alles') u.searchParams.delete('thema'); else u.searchParams.set('thema', slug(filter));
      const to = u.pathname + u.search + u.hash;
      if (u.href === w.location.href) return;
      const st = w.history.state && typeof w.history.state === 'object' ? { ...w.history.state } : w.history.state;
      if (st && st.as) st.as = to;
      /* Van het overzicht naar een thema: nieuwe stap in de geschiedenis, zodat 'terug' weer het overzicht toont */
      if (prevFilter === 'Alles' && filter !== 'Alles') w.history.pushState(st, '', to);
      else w.history.replaceState(st, '', to);
    } catch {}
  }

  /* Adres en weergave gelijk houden: terugknop van de browser, of het menu 'Catalogus' terwijl er een thema open staat */
  setInterval(() => {
    if (!cards.length) return;
    let wanted = '';
    try { wanted = new URLSearchParams(window.parent.location.search).get('thema') || ''; } catch { return; }
    const current = filter === 'Alles' ? '' : slug(filter);
    if (slug(wanted) === current) return;
    const target = wanted ? buttons.find(b => !b.hidden && slug(b.dataset.filter) === slug(wanted)) : null;
    const active = buttons.find(b => b.classList.contains('active'));
    syncing = true;
    try {
      if (target) target.click();
      else if (active) active.click();
    } finally { syncing = false; }
  }, 400);

  /* Het subthema-paneel komt direct onder de rij van de gekozen tegel. */
  function placePanel() {
    if (panel.hidden) return;
    const active = buttons.find(b => b.classList.contains('active'));
    if (!active) return;
    filters.appendChild(panel);                 // eerst uit het raster halen voor een zuivere meting
    const top = active.offsetTop;
    const lastInRow = buttons.filter(b => b.offsetParent && b.offsetTop === top).pop() || active;
    lastInRow.after(panel);
  }
  let resizeFrame;
  window.addEventListener('resize', () => {
    cancelAnimationFrame(resizeFrame);
    resizeFrame = requestAnimationFrame(placePanel);
  });

  panel.hidden = true;

  /* Soort kiezen: alles, documentaires of reeksen. Blijft behouden bij een ander thema. */
  formatButtons.forEach(button => button.addEventListener('click', () => {
    format = button.dataset.format;
    formatButtons.forEach(item => {
      const active = item === button;
      item.classList.toggle('active', active);
      item.setAttribute('aria-pressed', String(active));
    });
    applyFilter();
  }));

  /* Sorteren in het raster: A–Z (zonder lidwoord) of nieuwste eerst */
  sortButtons.forEach(button => button.addEventListener('click', () => {
    sort = button.dataset.sort;
    sortButtons.forEach(item => {
      const active = item === button;
      item.classList.toggle('active', active);
      item.setAttribute('aria-pressed', String(active));
    });
    applyFilter();
  }));

  /* Thema uit het adres, bv. …/pages/catalogus?thema=geschiedenis (vanaf de Welkom-pagina).
     De naam is de tegelnaam zonder hoofdletters en leestekens: kunst-cultuur, mens-samenleving, … */
  const slug = text => text.toLocaleLowerCase('nl').normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  function openThemeFromAddress() {
    let search = location.search;
    try { search = window.parent.location.search || search; } catch {}
    const wanted = new URLSearchParams(search).get('thema');
    if (!wanted) return;
    const button = buttons.find(b => !b.hidden && slug(b.dataset.filter) === slug(wanted));
    if (button && !button.classList.contains('active')) { syncing = true; try { button.click(); } finally { syncing = false; } }
    /* meteen naar het gekozen thema scrollen (tegel met daaronder de subthema's en titels) */
    if (button) setTimeout(() => button.scrollIntoView({ block:'start', behavior:'smooth' }), 350);
  }

  /* ---------- Opstarten ---------- */
  fetchAll().then(items => {
    buildPageMap(items);
    /* Opbouw in Revidd: reeks (series) → seizoenen (seasons) → afleveringen.
       Afleveringen tonen we niet los, en een seizoen dat bij een reeks hoort ook niet:
       de reeks krijgt één kaart met alle afleveringen van haar seizoenen. */
    const episodesBySeason = {};
    /* Een seizoen dat bij een reeks hoort (Revidd: series.seasons) niet apart tonen: de reeks toont het al */
    const seasonsInSeries = new Set(items.flatMap(x => (x.seasons || []).map(se => se.id)));
    items.filter(i => i.seasonId).forEach(ep => {
      (episodesBySeason[ep.seasonId] = episodesBySeason[ep.seasonId] || []).push(ep);
    });
    cards = items
      .filter(i => !i.seasonId && !i.seriesId && !seasonsInSeries.has(i.id) && !isHidden(i, items))
      .map(i => {
        const seasonIds = i.seasons
          ? [...i.seasons].sort((a, b) => (a.order || 0) - (b.order || 0)).map(season => season.id)
          : [i.id];
        const episodes = seasonIds.flatMap(id => episodesBySeason[id] || []);
        const firstSeason = i.seasons && items.find(x => x.id === seasonIds[0]);
        /* Tags van de seizoenen tellen mee voor de reeks (bv. een prijs op S1) */
        const seasonTags = i.seasons
          ? items.filter(x => seasonIds.includes(x.id)).flatMap(x => x.tags || [])
          : [];
        return renderCard({ ...i, tags:[...(i.tags || []), ...seasonTags] },
          episodes, seasonIds.length, firstSeason || i);
      })
      .sort(byTitle);
    /* Heeft een reekspagina een HOOFDTITEL, dan toont de catalogus alleen die kaart */
    const leadPages = new Set(cards.filter(c => c.dataset.page && c.dataset.lead).map(c => c.dataset.page));
    cards = cards.filter(c => !leadPages.has(c.dataset.page) || c.dataset.lead);
    /* Gratis, behalve wat met een prijs op de kaart staat */
    const free = cards.filter(card => card.dataset.free).length;
    intro.textContent = free === cards.length
      ? ' Alle verhalen zijn gratis te bekijken.'
      : ' Alle verhalen zijn gratis te bekijken, tenzij anders vermeld.';
    intro.hidden = free === 0;
    hideEmptyTiles();
    formatBar.hidden = false;
    choices = buildChoices();
    buildMenu();
    applyFilter();
    placePanel();
    openThemeFromAddress();
  }).catch(err => {
    console.error(err);
    empty.textContent = 'De verhalen konden niet geladen worden. Probeer het later opnieuw.';
    empty.hidden = false;
  });
})();
</script>

      'footer[class],footer{--color-mainText:#1a1a18;--color-placeholderText:#55534f;--color-border:#ddd6c8;background:#f7f4ee!important;color:#1a1a18!important;border-color:#ddd6c8!important}' +
      'footer [class],footer *{color:#55534f!important;border-color:#ddd6c8!important;background-color:transparent!important}' +

    style.textContent += '.MuiDrawer-paper div.cursor-pointer:has(> p.rd-menu__item:empty){display:none!important}div.fixed:has(img[alt=\"NavLogo\"]) div.bg-headerBg{background-color:#f7f4ee!important;border-color:rgba(26,26,24,.3)!important}div.fixed:has(img[alt=\"NavLogo\"]) .MuiIconButton-root svg{filter:none!important;color:#1a1a18!important;fill:#1a1a18!important}';
    host.head.append(style);

    /* Het logo linkt bij Revidd naar de oude standaard startpagina (/nl/home).
       Zolang deze pagina open staat, gaat een klik op het logo naar de landingspagina (/nl). */
    const view = host.defaultView;
    const onLogo = e => {
      /* Logo in de kop én in de footer van Revidd: dat is geen gewone link maar een klikbaar blok rond het logo */
      const t = e.target;
      const box = t.closest && t.closest('a, [class*="cursor-pointer"]');
      const isLogo = (t.matches && t.matches('img[alt="NavLogo"]')) || (box && box.querySelector('img[alt="NavLogo"]'));
      if(!isLogo) return;
      e.preventDefault();
      e.stopPropagation();
      const lang = view.location.pathname.split('/')[1] || 'nl';
      view.location.href = '/' + lang;
    };
    host.addEventListener('click', onLogo, true);

    const cleanup = () => { style.remove(); host.removeEventListener('click', onLogo, true); };
    window.addEventListener('pagehide', cleanup);
    window.addEventListener('unload', cleanup);
  }

  /* Hoeveel van de kop ligt over het kader? Dat stuk schuiven we de inhoud omlaag. */
  const root = document.querySelector('.bagage-catalogus');
  const frame = window.frameElement;
  if(!root || !frame) return;
  const measure = () => {
    const logo = host.querySelector('img[alt="NavLogo"]');
    const header = logo && logo.closest('.fixed');
    if(!header) return;
    const overlap = Math.max(0, Math.round(((el) => { const v = el.ownerDocument.defaultView; let st = (el.ownerDocument.scrollingElement || el.ownerDocument.documentElement).scrollTop; for (let p = frame.parentElement; p && p !== el.ownerDocument.body; p = p.parentElement) { const oy = v.getComputedStyle(p).overflowY; if (oy !== 'visible' && oy !== 'clip' && p.scrollHeight > p.clientHeight + 1) { st = p.scrollTop; break; } } const b = el.getBoundingClientRect().bottom; if (st <= 2 || !v.__bagageHeaderMax) v.__bagageHeaderMax = Math.max(v.__bagageHeaderMax || 0, b); return v.__bagageHeaderMax - (frame.getBoundingClientRect().top + st); })(header)));
    root.style.setProperty('--revidd-header', overlap + 'px');
  };
  measure();
  window.addEventListener('load', measure);
  window.parent.addEventListener('resize', measure);
  setTimeout(measure, 1500);                    // logo en kop laden soms iets later
})();
